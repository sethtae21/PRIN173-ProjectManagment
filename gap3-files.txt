def _guest_index_key(session_key):
    return f'guest_rating_keys:{session_key}'


def _legacy_guest_keys(request):
    """Read-only lookup of LEGACY guest cache keys so they can be DELETED.
    Nothing is ever written to the cache for a guest (FR-1.9)."""
    session_key = request.session.session_key
    if not session_key:
        return []
    index_key = _guest_index_key(session_key)
    return [*(cache.get(index_key) or []), index_key]


def _is_shopper(user):
    return getattr(user, 'role', None) == 'shopper'


class RatingSubmitView(APIView):
    """
    POST /ratings/ â€” REGISTERED SHOPPERS ONLY, STAR-ONLY (1..5 integer).

    * Guest  -> 403 + signup/login prompt; NOTHING cached or persisted.
    * Seller -> 403 (ratings are a shopper surface).
    * Shopper-> upsert the (user, target_type, target_id, source) row's score.

    permission_classes stays AllowAny on purpose so guests get a clean 403
    (DRF IsAuthenticated would answer 401, which the frontend token lifecycle
    interprets as a session reset).
    """
    permission_classes = [AllowAny]

    def post(self, request):
        if not request.user.is_authenticated:
            logger.info("Guest rating attempt refused (403): no rating rights.")
            return Response({
                'detail': 'Registration is required to rate items.',
                'prompt': 'signup_login',
                'persisted': False,
            }, status=status.HTTP_403_FORBIDDEN)

        if not _is_shopper(request.user):
            return Response(
                {'detail': SHOPPER_ONLY_DETAIL, 'persisted': False},
                status=status.HTTP_403_FORBIDDEN,
            )

        serializer = RatingSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        vd = serializer.validated_data
        source = vd['source']
        if source not in VALID_SOURCES:
            return Response(
                {'detail': f'Invalid source "{source}". '
                           f'Allowed: {sorted(VALID_SOURCES)}'},
                status=status.HTTP_400_BAD_REQUEST,
            )

        rating, created = Rating.objects.get_or_create(
            user=request.user,
            target_type=vd['target_type'],
            target_id=vd['target_id'],
            source=source,
            defaults={'score': vd['score']},
        )
        if not created:
            rating.score = vd['score']
            rating.save(update_fields=['score', 'updated_at'])

        return Response({
            'persisted': True,
            'created': created,
            'rating': RatingSerializer(rating).data,
        }, status=status.HTTP_201_CREATED if created else status.HTTP_200_OK)


class RatingSummaryView(APIView):
    """
    GET /ratings/summary/?target_type=&target_id=[&source=]
    Public STAR aggregate. 4-5 stars = approval (approval_count).
    """
    permission_classes = [AllowAny]

    def get(self, request):
        target_type = request.query_params.get('target_type')
        target_id = request.query_params.get('target_id')
        if not target_type or not target_id:
            return Response({'error': 'target_type and target_id are required'},
                            status=status.HTTP_400_BAD_REQUEST)

        source = request.query_params.get('source', '')
        if source and source not in VALID_SOURCES:
            return Response(
                {'error': f'Invalid source "{source}". '
                          f'Allowed: {sorted(VALID_SOURCES)}'},
                status=status.HTTP_400_BAD_REQUEST,
            )

        qs = Rating.objects.filter(target_type=target_type, target_id=target_id)
        if source:
            qs = qs.filter(source=source)

        scores = [s for s in qs.values_list('score', flat=True) if s is not None]
        score_count = len(scores)
        average = round(sum(scores) / score_count, 2) if score_count else 0.0

        star_counts = {str(i): 0 for i in range(1, 6)}
        for row in qs.values('score').annotate(count=Count('id')):
            if row['score'] is not None:
                star_counts[str(row['score'])] = row['count']

        approval_count = sum(1 for s in scores if s >= APPROVAL_MIN_SCORE)

        return Response(RatingSummarySerializer({
            'target_type': target_type,
            'target_id': target_id,
            'source': source or 'all',
            'score_count': score_count,
            'average': average,
            'star_counts': star_counts,
            'approval_count': approval_count,
        }).data)


class GuestSessionClearView(APIView):
    """POST /ratings/guest-session/clear/ â€” deletes any LEGACY guest cache
    aggregates and flushes the guest session (FR-1.9 housekeeping).
    Never writes anything."""
    permission_classes = [AllowAny]

    def post(self, request):
        if request.user.is_authenticated:
            return Response({'cleared': False}, status=status.HTTP_200_OK)
        keys = _legacy_guest_keys(request)
        if keys:
            cache.delete_many(keys)
        if request.session.session_key:
            request.session.flush()
        return Response({'cleared': True}, status=status.HTTP_200_OK)

class RatingDetailView(APIView):
    """PATCH / DELETE /ratings/<id>/ â€” owner-only, shopper-only.

    CHANGED (audit fix #8a): delete now also requires `_is_shopper`. Previously
    create required shopper but delete only required ownership, which left a
    symmetry gap (a seller couldn't create but could delete their own, and a
    seller who somehow got a row into the table could remove it).
    """
    permission_classes = [IsAuthenticated]

    def _get_own(self, request, pk):
        return Rating.objects.filter(user=request.user, id=pk).first()

    def patch(self, request, pk):
        rating = self._get_own(request, pk)
        if rating is None:
            return Response({'error': 'Rating not found'}, status=status.HTTP_404_NOT_FOUND)
        if not _is_shopper(request.user):
            return Response({'detail': SHOPPER_ONLY_DETAIL}, status=status.HTTP_403_FORBIDDEN)
        serializer = RatingSerializer(rating, data=request.data, partial=True)
        serializer.is_valid(raise_exception=True)
        if 'score' in serializer.validated_data:
            rating.score = serializer.validated_data['score']
            rating.save(update_fields=['score', 'updated_at'])
        return Response(RatingSerializer(rating).data)

    def delete(self, request, pk):
        if not _is_shopper(request.user):
            return Response({'detail': SHOPPER_ONLY_DETAIL}, status=status.HTTP_403_FORBIDDEN)
        rating = self._get_own(request, pk)
        if rating is None:
            return Response({'error': 'Rating not found'}, status=status.HTTP_404_NOT_FOUND)
        rating.delete()
        return Response(status=status.HTTP_204_NO_CONTENT)


class UATSessionCreateView(APIView):
    """POST /ratings/uat/session/ â€” start a tester session.

    FIX (OBJ-01 join): persist product_id from the payload. The OBJ-01 KPI
    joins fit-visualization ratings to sessions by (user, product).

    CHANGED (audit fix #8b): REGISTERED SHOPPERS ONLY. A seller logging a UAT
    session would contaminate the OBJ-01 / OBJ-05 / OBJ-06 evidence denominators.
    """
    permission_classes = [IsAuthenticated]

    def post(self, request):
        if not _is_shopper(request.user):
            return Response({'detail': SHOPPER_ONLY_DETAIL}, status=status.HTTP_403_FORBIDDEN)

        role = request.data.get('role', request.user.role)
        if role not in {'shopper', 'seller'}:
            return Response({'detail': 'Invalid UAT role.'}, status=status.HTTP_400_BAD_REQUEST)
        if request.user.role != role:
            return Response({'detail': 'Role mismatch for this tester session.'}, status=status.HTTP_403_FORBIDDEN)

        product_id = str(request.data.get('product_id') or '').strip()
        session = UATSession.objects.create(
            user=request.user, role=role, status='active', product_id=product_id)
        return Response({
            'session_id': session.session_id,
            'status': session.status,
            'role': session.role,
            'product_id': session.product_id,
        }, status=status.HTTP_201_CREATED)


class UATResponseSubmitView(APIView):
    """POST /ratings/uat/response/ â€” save one survey answer.

    `value == 0` is LEGAL (style_discovery "No"). Fractions are refused, not
    truncated. Unknown question keys are rejected against KNOWN_QUESTION_KEYS.
    Sessions stay 'active' so the server-side commerce hook can still record
    cart_created / checkout_completed.

    CHANGED (audit fix #8b): REGISTERED SHOPPERS ONLY. A seller logging a UAT
    response would contaminate the charter evidence paired-answer calculations.
    """
    permission_classes = [IsAuthenticated]

    def post(self, request):
        if not _is_shopper(request.user):
            return Response({'detail': SHOPPER_ONLY_DETAIL}, status=status.HTTP_403_FORBIDDEN)

        session_id = request.data.get('session_id')
        if not session_id:
            return Response({'detail': 'session_id is required.'}, status=status.HTTP_400_BAD_REQUEST)

        session = UATSession.objects.filter(session_id=session_id, user=request.user).first()
        if session is None:
            return Response({'detail': 'UAT session not found or not owned by this account.'}, status=status.HTTP_404_NOT_FOUND)

        product_id = str(request.data.get('product_id') or '').strip()
        question_key = str(request.data.get('question_key') or request.data.get('key') or '').strip()
        phase = str(request.data.get('phase') or 'before').strip()
        if not question_key:
            return Response({'detail': 'question_key is required.'}, status=status.HTTP_400_BAD_REQUEST)
        if not product_id:
            return Response({'detail': 'product_id is required.'}, status=status.HTTP_400_BAD_REQUEST)
        if question_key not in KNOWN_QUESTION_KEYS:
            return Response(
                {'detail': f"Unknown UAT question '{question_key}'. "
                           f'Allowed: {", ".join(sorted(KNOWN_QUESTION_KEYS))}.'},
                status=status.HTTP_400_BAD_REQUEST,
            )

        value = request.data.get('value')
        if value is None or value == '':
            return Response({'detail': 'value is required.'}, status=status.HTTP_400_BAD_REQUEST)

        if isinstance(value, bool):
            return Response({'detail': 'value must be a whole number.'}, status=status.HTTP_400_BAD_REQUEST)
        if isinstance(value, float):
            if not value.is_integer():
                return Response({'detail': 'value must be a whole number.'}, status=status.HTTP_400_BAD_REQUEST)
            numeric_value = int(value)
        elif isinstance(value, int):
            numeric_value = value
        elif isinstance(value, str):
            if '.' in value:
                return Response({'detail': 'value must be a whole number.'}, status=status.HTTP_400_BAD_REQUEST)
            try:
                numeric_value = int(value)
            except ValueError:
                return Response({'detail': 'value must be a whole number.'}, status=status.HTTP_400_BAD_REQUEST)
        else:
            return Response({'detail': 'value must be a whole number.'}, status=status.HTTP_400_BAD_REQUEST)

        if not UAT_VALUE_MIN <= numeric_value <= UAT_VALUE_MAX:
            return Response(
                {'detail': f'value must be between {UAT_VALUE_MIN} and {UAT_VALUE_MAX}.'},
                status=status.HTTP_400_BAD_REQUEST,
            )

        obj, created = UATResponse.objects.update_or_create(
            session=session,
            product_id=product_id,
            question_key=question_key,
            phase=phase,
            defaults={'value': numeric_value},
        )
        return Response({
            'created': created,
            'response': UATResponseSerializer(obj).data,
        }, status=status.HTTP_201_CREATED if created else status.HTTP_200_OK)

class Rating(models.Model):
    """
    In-app STAR rating (SRS FR-7.5 / Scope Criteria 9-11).

    STAR-ONLY BASELINE:
      * score is a REQUIRED integer 1..5. 4-5 = Approval, 1-3 = Not approval.
      * There is NO vote / like / dislike dimension anywhere (model, API,
        summary, KPI). Legacy vote-only rows were dropped by migration 0003.
      * REGISTERED SHOPPERS ONLY. Guests are refused server-side (403) and
        nothing is cached or persisted for them (FR-1.9). Sellers are refused
        too â€” ratings are a shopper surface.

    `source` tags the UI surface so UAT KPI reports separate cleanly:
      * recommendation_panel -> OBJ-05 / Criterion 10 (per recommended item)
      * fit_visualization    -> OBJ-01 / Criterion 9  (on-avatar + checkout
                                fit-accuracy)
      * product_page         -> product review (context only)
      * order_history        -> post-purchase (context only)
    """
    TARGET_CHOICES = [('item', 'Catalog Item'), ('fit', 'Fit Visualization')]
    SOURCE_CHOICES = [
        ('recommendation_panel', 'Recommendation Panel'),
        ('fit_visualization', 'Fit Visualization'),
        ('product_page', 'Product Page'),
        ('order_history', 'Order History'),
    ]

    user = models.ForeignKey(
        settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name='ratings')
    target_type = models.CharField(max_length=10, choices=TARGET_CHOICES)
    target_id = models.CharField(max_length=50)  # ObjectId stored as string

    # REQUIRED, non-null, strict 1..5 stars.
    score = models.IntegerField(
        validators=[MinValueValidator(1), MaxValueValidator(5)])

    source = models.CharField(max_length=32, choices=SOURCE_CHOICES)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = 'accounts_rating'
        unique_together = ('user', 'target_type', 'target_id', 'source')
        indexes = [
            models.Index(fields=['target_type', 'target_id']),
            models.Index(fields=['source']),
        ]
        ordering = ['-updated_at']

    def __str__(self):
        return (
            f"{self.user} -> {self.target_type}:{self.target_id} "
            f"[{self.score}â˜…] ({self.source})"
        )

from rest_framework import serializers

class StrictStarField(serializers.IntegerField):
    """
    STRICT 1..5 integer stars (SRS FR-7.5 star-only baseline).

    DRF's IntegerField happily coerces "5", 5.0 and True into 5. The baseline
    requires an actual integer, so every non-int (including bool, which is an
    int subclass in Python) is rejected with 400 instead of being coerced.
    """

    def to_internal_value(self, data):
        if isinstance(data, bool) or not isinstance(data, int):
            self.fail('invalid')
        return super().to_internal_value(data)


class RatingSerializer(serializers.ModelSerializer):
    """
    STAR-ONLY payload: { target_type, target_id, score (int 1..5), source }.
    `score` and `source` are REQUIRED. There is no vote/like/dislike field.

    HARDENING (defense-in-depth): legacy `vote` / `like` / `dislike` keys in
    the request body are EXPLICITLY REJECTED (400), not silently dropped.
    Without this guard, a stale client sending `{"score": 5, "vote": "like"}`
    would still save the score and the vote would vanish â€” masking the
    client-side drift from the terminal.
    """
    id = serializers.CharField(read_only=True)
    user = serializers.CharField(read_only=True, source='user_id')

    score = StrictStarField(min_value=1, max_value=5)
    source = serializers.ChoiceField(choices=Rating.SOURCE_CHOICES)

    class Meta:
        model = Rating
        fields = ['id', 'user', 'target_type', 'target_id', 'score',
                  'source', 'created_at', 'updated_at']
        read_only_fields = ['user', 'created_at', 'updated_at']

    def to_internal_value(self, data):
        if hasattr(data, 'keys'):
            legacy = {'vote', 'like', 'dislike'}.intersection(data.keys())
            if legacy:
                raise serializers.ValidationError({
                    key: 'Like/dislike voting is no longer supported; submit a 1-5 star score instead.'
                    for key in sorted(legacy)
                })
        return super().to_internal_value(data)

class ArchiveEligibilityMathTests(TestCase):

    def test_guest_submit_rejected_and_not_persisted(self):

        to the DB, and the cache is never written to (FR-1.9 ephemeral guests)."""
        from unittest.mock import patch
        from django.core.cache import cache

        with patch.object(cache, 'set') as cache_set, \
             patch.object(cache, 'add') as cache_add:
            res = self.client.post('/ratings/', {**self.t, 'score': 5}, format='json')
        
        self.assertEqual(res.status_code, 403)
        self.assertEqual(res.data['prompt'], 'signup_login')
        self.assertFalse(res.data['persisted'])
        self.assertEqual(Rating.objects.count(), 0)
        self.assertEqual(Session.objects.count(), 0)
        self.assertNotIn('session_aggregate', res.data)
        # Cache-spy: nothing was ever written for a guest
        cache_set.assert_not_called()
        cache_add.assert_not_called()

    def test_owner_only_update_delete(self):

        self.client.force_authenticate(self.other)

        self.assertEqual(self.client.delete(f'/ratings/{rid}/').status_code, 404)
        self.client.force_authenticate(self.user)
        self.assertEqual(self.client.delete(f'/ratings/{rid}/').status_code, 204)

    def test_seller_cannot_delete_any_rating(self):
        """Shopper-only RBAC closes the create/delete asymmetry.

        The patch guard on `RatingDetailView.delete` ensures a seller can
        neither create a rating nor remove another user's. The row survives
        the attempted delete.
        """
        self.client.force_authenticate(self.user)
        rid = self.client.post(
            '/ratings/', {**self.t, 'score': 3}, format='json').data['rating']['id']

        self.client.force_authenticate(self.seller)
        self.assertEqual(
            self.client.delete(f'/ratings/{rid}/').status_code,
            403,
            "seller could delete another user's rating",
        )
        self.assertTrue(Rating.objects.filter(id=rid).exists())

class UATEndpointRBACTests(TestCase):
    """OBJ-01 / OBJ-05 evidence integrity: UAT survey writes are shopper-only.

    A seller logging a UAT session or response would contaminate the paired
    pre/post evidence (OBJ-01 / OBJ-06) and the recommendation-approval
    denominator (OBJ-05). Both endpoints must refuse sellers with 403;
    shoppers still succeed.
    """

    def setUp(self):
        self.client = APIClient()
        self.shopper = User.objects.create_user(
            email='uat-shopper@test.com', password='Test123!@#',
            full_name='UAT Shopper', role='shopper',
        )
        self.seller = User.objects.create_user(
            email='uat-seller@test.com', password='Test123!@#',
            full_name='UAT Seller', role='seller', store_name='UAT Store',
        )
        self.session_payload = {
            'role': 'shopper',
            'product_id': '6ab52dc83c8d812f8b7ee414',
        }

    def test_uat_session_shopper_allowed(self):
        self.client.force_authenticate(self.shopper)
        res = self.client.post('/ratings/uat/session/', self.session_payload, format='json')
        self.assertEqual(res.status_code, 201)

    def test_uat_session_seller_rejected(self):
        self.client.force_authenticate(self.seller)
        res = self.client.post('/ratings/uat/session/', self.session_payload, format='json')
        self.assertEqual(res.status_code, 403)
        from .models import UATSession
        self.assertEqual(UATSession.objects.count(), 0)

    def test_uat_response_seller_rejected(self):
        """A seller must not be able to log survey answers into any session."""
        # Create a session as the shopper first so there's a valid target.
        self.client.force_authenticate(self.shopper)
        session = self.client.post(
            '/ratings/uat/session/', self.session_payload, format='json').data

        # Now try to log a response as the seller.
        self.client.force_authenticate(self.seller)
        res = self.client.post(
            '/ratings/uat/response/',
            {
                'session_id': session['session_id'],
                'product_id': self.session_payload['product_id'],
                'phase': 'before',
                'question_key': 'fit_confidence',
                'value': 3,
            },
            format='json',
        )
        self.assertEqual(res.status_code, 403)
        from .models import UATResponse
        self.assertEqual(UATResponse.objects.count(), 0)

from rest_framework import permissions

class IsSeller(permissions.BasePermission):
    """
    Allows access only to authenticated users with the 'seller' role.
    """
    def has_permission(self, request, view):
        return (
            request.user and 
            request.user.is_authenticated and 
            getattr(request.user, 'role', None) == 'seller'
        )
