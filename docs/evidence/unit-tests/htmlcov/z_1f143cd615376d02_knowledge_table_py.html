1""" 

2FitFusion AI - color-theory & size-range knowledge table (rule-based engine). 

3 

4Traceability: SRS FR-7.1, FR-7.2 | Scope Baseline WBS 1.3.5 | Charter OBJ-05. 

5Nature: deterministic, code-encoded knowledge (symbolic/heuristic AI). 

6No machine learning - every rule is human-readable and auditable. 

7 

8Weight tuning: before the baseline freeze, bump WEIGHTS_VERSION and record the 

9change + rationale in the tuning log (KAN-107). AFTER the freeze, weight changes 

10are baselined changes and must go through the CCB (Charter Sec. 8). 

11 

12Size-policy revision (integration with female/male avatar fix + size update): 

13* Petite / average / tall height categories REMOVED. Height is a single 

14 range: 120.0-218.0 cm. The FR-7.2 contrast rule now uses a deterministic 

15 midpoint split of that range (below / above 169.0 cm) - no named bands. 

16* Body build range added: 86.0-112.0 cm. 

17* Clothing sizes: S, M, L, XL, 2XL, 3XL, 4XL. The clothing size table is 

18 REFERENCE-ONLY (displayed on product details in the "AVAILABLE SIZE" 

19 block as e.g. "M (90-94 cm)" + "Standard size chart reference."): 

20 pure cm, no inches, no regional equivalents. Never scored, never stored. 

21* Footwear PERMANENTLY REMOVED from the catalog: the regional shoe-size 

22 table (EU/US/UK), its validation branches and its display helpers are 

23 gone. Regional codes like "US 9" / "EU 40" are now rejected everywhere. 

24* Body-proportion TOTAL OUTPUT = recommended size + average cm, where 

25 average = (height_cm + body_build_cm) / 2. 

26"""

60# --- Body ranges (single source of truth) ------------------------------------ 

61HEIGHT_RANGE = (120.0, 218.0) # cm, inclusive 

62BODY_BUILD_RANGE = (86.0, 112.0) # cm, inclusive 

63HEIGHT_MIDPOINT = (HEIGHT_RANGE[0] + HEIGHT_RANGE[1]) / 2.0 # 169.0 cm

65# --- FR-7.2 rule 2: height-appropriate contrast ------------------------------ 

66# No named height bands anymore: a deterministic midpoint split of 

67# HEIGHT_RANGE. Shorter stature keeps the silhouette unbroken (low/medium 

68# contrast); taller stature can carry silhouette-breaking contrast. 

69CONTRAST_BY_HEIGHT_ZONE = { 

70 "below-midpoint": {"low", "medium"}, 

71 "above-midpoint": {"medium", "high"},

75def clamp_range(value, bounds): 

76 """Clamp a numeric value into bounds; invalid input falls back to min.""" 

77 try: 

78 number = float(value) 

79 except (TypeError, ValueError): 

80 number = bounds[0] 

81 return min(bounds[1], max(bounds[0], number))

84def height_zone(height_cm) -> str: 

85 """Deterministic height -> zone mapping (helper for FR-7.2 rule 2).""" 

86 height = clamp_range(height_cm, HEIGHT_RANGE) 

87 return "below-midpoint" if height < HEIGHT_MIDPOINT else "above-midpoint"

110# --- Size range table (REFERENCE-ONLY, expert-encoded) ----------------------- 

111# Clothing sizes: displayed on product details as "M (90-94 cm)". 

112# Pure CM - no inches, no regional equivalents. Never scored, never stored. 

113# Sellers input size labels ONLY (see validate_size_label); cm entries are 

114# rejected at upload/edit time. 

115# Bands are half-open [lo, hi) except the last band, which includes 112 cm, 

116# so the whole 86-112 cm body-build range maps to exactly one size. 

117SIZE_RANGE_TABLE = { 

118 'S': {'cm': (86, 90)}, 

119 'M': {'cm': (90, 94)}, 

120 'L': {'cm': (94, 98)}, 

121 'XL': {'cm': (98, 102)}, 

122 '2XL': {'cm': (102, 106)}, 

123 '3XL': {'cm': (106, 110)}, 

124 '4XL': {'cm': (110, 112)}, 

125}

127SIZE_OPTIONS = ('S', 'M', 'L', 'XL', '2XL', '3XL', '4XL') 

128CLOTHING_SIZE_OPTIONS = SIZE_OPTIONS # legacy alias (same policy for clothing)

130# --- Size label validation (CSV upload + listing edits) ---------------------- 

131# Every category (tops/bottoms/dresses/outerwear): label must be one of S..4XL. 

132# Any cm/measured entry is rejected. Footwear was permanently removed, so 

133# regional shoe codes (US/EU/UK) are no longer accepted for any category. 

134_MEASURED_RE = re.compile( 

135 r'\d+(?:\.\d+)?\s*(?:inch|inches|in|cm|centimeter|centimeters|mm|"|″)', 

136 re.IGNORECASE, 

137)

140def validate_size_label(size_text, category=''): 

141 """ 

142 Deterministic size-format rule. Returns (is_valid, error_message_or_None). 

143 

144 All categories: label must be S, M, L, XL, 2XL, 3XL, or 4XL. 

145 Any cm/measured entry is rejected. Regional shoe codes are rejected 

146 (footwear was permanently removed from the catalog). 

147 The `category` parameter is kept for call-site compatibility only. 

148 Empty values pass here - presence is enforced by the required-field rule. 

149 """ 

150 raw = str(size_text or '').strip() 

151 if not raw: 

152 return True, None 

153 

154 key = raw.upper() 

155 

156 if _MEASURED_RE.search(raw): 

157 return False, ( 

158 f"Invalid size '{raw}': measured values with units (e.g., cm) are not accepted. " 

159 f"Enter a standard size label ({', '.join(SIZE_OPTIONS)})." 

160 ) 

161 

162 if key in SIZE_OPTIONS: 

163 return True, None 

164 return False, ( 

165 f"Invalid size '{raw}'. Use standard clothing sizes only: " 

166 f"{', '.join(SIZE_OPTIONS)}. Do not enter cm measurements; " 

167 f"the product page shows the cm range automatically." 

168 )

171def get_size_ranges(size_text): 

172 """ 

173 Convert a size label to its display ranges - the exact content of the 

174 product-details "AVAILABLE SIZE" block. 

175 

176 Clothing (S..4XL): {'label', 'cm', 'regions': None} where 'cm' renders as 

177 "(90-94 cm)" so the page shows e.g. "M (90-94 cm)". The 'regions' key is 

178 kept (always None) for payload-shape compatibility after the footwear 

179 removal. Returns None for unknown sizes. Display-only: never scored, 

180 never stored. 

181 """ 

182 if not size_text: 

183 return None 

184 raw_key = str(size_text).strip().upper() 

185 

186 entry = SIZE_RANGE_TABLE.get(raw_key) 

187 if entry: 

188 return { 

189 'label': raw_key, 

190 'cm': f"({entry['cm'][0]}-{entry['cm'][1]} cm)", 

191 'regions': None, 

192 } 

193 return None

196def size_for_body_build(body_build_cm): 

197 """Map a body-build measurement (86-112 cm) to its S..4XL size label.""" 

198 value = clamp_range(body_build_cm, BODY_BUILD_RANGE) 

199 last = SIZE_OPTIONS[-1] 

200 for label in SIZE_OPTIONS: 

201 lo, hi = SIZE_RANGE_TABLE[label]['cm'] 

202 if label == last: 

203 if lo <= value <= hi: 

204 return label 

205 elif lo <= value < hi: 

206 return label 

207 return last

210def body_proportion(height_cm, body_build_cm): 

211 """ 

212 TOTAL body-proportion output: {'size': <S..4XL>, 'average': <cm>}. 

213 

214 size = SIZE_RANGE_TABLE lookup on the clamped body build (86-112 cm). 

215 average = (height_cm + body_build_cm) / 2, clamped to 120-218 / 86-112 

216 first, rounded to one decimal. 

217 """ 

218 height = clamp_range(height_cm, HEIGHT_RANGE) 

219 build = clamp_range(body_build_cm, BODY_BUILD_RANGE) 

220 return { 

221 'size': size_for_body_build(build), 

222 'average': round((height + build) / 2.0, 1), 

223 }

239def infer_contrast(color_description: str) -> str: 

240 """Keyword -> contrast level; first match wins, else 'medium'.""" 

241 text = (color_description or "").lower() 

242 for level, keywords in CONTRAST_KEYWORDS.items(): 

243 if any(k in text for k in keywords): 

244 return level 

245 return DEFAULT_CONTRAST
