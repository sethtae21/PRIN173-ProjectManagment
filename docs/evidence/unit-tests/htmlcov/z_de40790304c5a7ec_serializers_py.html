28 def validate_size(self, value): 

29 """Sellers may only store S, M, L, XL, 2XL, 3XL, 4XL labels. Inch/cm 

30 entries and regional shoe codes are rejected on PUT/PATCH exactly as 

31 on CSV upload (footwear was permanently removed).""" 

32 category = '' 

33 if isinstance(getattr(self, 'initial_data', None), dict): 

34 category = str(self.initial_data.get('category') or '').strip().lower() 

35 if not category and self.instance is not None: 

36 category = str(getattr(self.instance, 'category', '') or '').strip().lower() 

37 is_valid, error = validate_size_label(value, category) 

38 if not is_valid: 

39 raise serializers.ValidationError(error) 

40 return value
