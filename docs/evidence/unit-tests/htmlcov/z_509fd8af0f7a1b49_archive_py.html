1""" 

2accounts/archive.py 

3The cold-relocation sweep (FR-1.4 / RA 10173 end-of-retention). 

4 

5ELIGIBILITY ALGORITHM (the forward form you asked to make explicit): 

6 due_date = archived_at + ARCHIVE_GRACE_DAYS # archive_due_date() 

7 relocatable when due_date < now # (or archived_at is NULL 

8 # = legacy row = due now) 

9The original code expressed the same rule backwards 

10(archived_at < now - ARCHIVE_GRACE_DAYS); both are identical, but the forward 

11form is what `describe_archive_schedule()` prints so you can SEE a row's due 

12date and the days remaining instead of guessing why it did or didn't move. 

13NOTE: the queryset uses a STRICT `<` cutoff, so a row exactly at its due 

14instant becomes eligible on the next sweep tick, not the same microsecond. 

15 

16CORRECTED (audit-truth wording — behaviour unchanged): 

17`cold_moved_at` stores the SCHEDULED window date computed by 

18`ArchivedUser.from_user()` (archived_at + ARCHIVE_GRACE_DAYS), NOT the 

19wall-clock instant of relocation. Legacy rows with no archived_at fall back to 

20`timezone.now()`. This is the documented trade-off recorded on the 

21ArchivedUser model: the field audits the SCHEDULE, so "when did the data 

22actually leave accounts_user" is intentionally not recorded there. Eligibility 

23is always computed live from ARCHIVE_GRACE_DAYS, so this stored value never 

24drives the sweep. If a true relocation timestamp is required for the RA 10173 

25audit trail, add a separate field (e.g. relocated_at) rather than overloading 

26this one. 

27 

28PRIVACY (FR-1.4): the cold copy written in step 1 is PSEUDONYMOUS — no email, 

29name, contact/address, store identity, or password hash is retained. Legacy 

30intact cold documents are scrubbed by migration 0012 and the repeatable 

31`pseudonymize_archived_users` command (or `prune_archived_users 

32--sanitize-existing`). 

33 

34Moves accounts that have sat in the 'archive' grace window for >= ARCHIVE_GRACE_DAYS 

35out of accounts_user and into the accounts_archiveduser cold collection, then 

36detaches their linked data using the EXACT cascade your original hard-delete used 

37(cart fix-up loop + order pseudonymize loop + user.delete()), so the live 

38collections end up with zero dangling references. 

39 

40Why reuse the old cascade instead of inventing a new one: that code path was 

41already proven crash-free by your test suite; replicating it verbatim at the end 

42of the window is the lowest-risk way to guarantee no view ever navigates a dead 

43order.user / item.seller id. The ONLY differences from the old immediate delete 

44are (a) the grace-window delay and (b) the pseudonymized cold tombstone written first. 

45 

46Called from two places: 

47 * accounts/management/commands/prune_archived_users.py (production / cron / --force / --preview) 

48 * LoginView.post (lazy, best-effort, never blocks login) (zero-config dev) 

49Models are imported lazily inside the functions to avoid any import cycle. 

50 

51Proof: accounts/tests_archive_retention.py (eligibility math, preview, cold-copy 

52pseudonymization + idempotency, erasure cascade, order pseudonymization, 

53seller-cart detach, sweep resilience, legacy backfill). 

54""" 

55import logging 

56import math 

57from datetime import datetime, timedelta, timezone as _StdTZ 

58 

59from django.db.models import Q 

60from django.utils import timezone 

61 

62logger = logging.getLogger(__name__) 

63 

64# Grace window before a closed profile is relocated to cold storage. 

65# Baselined value: SRS FR-1.4 / Charter / E2E-10 specify 90 days. Every 

66# eligibility calculation reads this constant, so changing it here changes the 

67# whole schedule consistently (demo/UAT overrides go through --days / --force). 

68ARCHIVE_GRACE_DAYS = 90 

69 

70# Aware epoch used ONLY as a sort sentinel for legacy (NULL due) rows in the 

71# preview; never written to the DB. Django 5+ removed django.utils.timezone.utc, 

72# so the aware UTC comes from the stdlib datetime (works on every Django version). 

73_EPOCH = datetime(1970, 1, 1, tzinfo=_StdTZ.utc) 

74 

75 

76def archive_due_date(user, days: int = ARCHIVE_GRACE_DAYS): 

77 """Forward grace-window calculation: archived_at + `days`. 

78 

79 Returns None for legacy rows with no archived_at stamp, which callers 

80 interpret as "due immediately". This is the single source of truth for the 

81 due date; the User.archive_due_date property mirrors it. 

82 """ 

83 archived_at = getattr(user, 'archived_at', None) 

84 if archived_at is None: 

85 return None 

86 return archived_at + timedelta(days=days)

89def _eligibility_queryset(days: int = ARCHIVE_GRACE_DAYS, force: bool = False): 

90 """Rows in accounts_user that may be relocated right now. 

91 

92 force=True ignores the clock entirely (every status='archive' row) — this 

93 is the instant terminal move. Otherwise: archived longer than `days`, OR 

94 never timestamped (legacy isnull branch). 

95 """ 

96 from accounts.models import User 

97 

98 base = User.objects.filter(status='archive') 

99 if force: 

100 return base 

101 cutoff = timezone.now() - timedelta(days=days) 

102 return base.filter( 

103 Q(archived_at__lt=cutoff) | Q(archived_at__isnull=True) 

104 )

107def describe_archive_schedule(days: int = ARCHIVE_GRACE_DAYS, force: bool = False) -> list: 

108 """Preview, per archived row: due date, days remaining, eligible-now flag. 

109 

110 Iterates EVERY status='archive' row (not just eligible ones) so you can see 

111 a same-day archive listed as "due <date> -> in N day(s) -> NOT eligible", 

112 proving the algorithm. Sorted soonest-due first; legacy NULL-due rows float 

113 to the top. 

114 """ 

115 from accounts.models import User 

116 

117 now = timezone.now() 

118 rows = [] 

119 for user in User.objects.filter(status='archive'): 

120 due = archive_due_date(user, days=days) 

121 if due is None: 

122 eligible = True 

123 remaining_label = "immediate (legacy: no archived_at)" 

124 else: 

125 secs = (due - now).total_seconds() 

126 if secs <= 0: 

127 eligible = True 

128 remaining_label = "OVERDUE -> eligible now" 

129 else: 

130 eligible = False 

131 remaining_label = f"in {math.ceil(secs / 86400)} day(s)" 

132 if force: 

133 eligible = True 

134 if due is not None and (due - now).total_seconds() > 0: 

135 remaining_label += " (forced)" 

136 rows.append({ 

137 "id": str(user.pk), 

138 "email": user.email, 

139 "full_name": user.full_name, 

140 "archived_at": user.archived_at, 

141 "due_date": due, 

142 "eligible": eligible, 

143 "remaining": remaining_label, 

144 }) 

145 

146 rows.sort(key=lambda r: (r["due_date"] is not None, r["due_date"] or _EPOCH)) 

147 return rows

150def prune_expired_archived_users(days: int = ARCHIVE_GRACE_DAYS, force: bool = False) -> int: 

151 """Relocate eligible archived users to accounts_archiveduser. 

152 

153 force=True bypasses the grace-window clock (instant move for demos/UAT). 

154 Each user is handled independently; a failure on one is logged and skipped 

155 so a single bad row can never stall the whole sweep. The pseudonymized cold 

156 tombstone is written BEFORE the live row is deleted, so even without a 

157 transaction the worst case is a duplicate cold row, never data loss. 

158 Returns users processed. 

159 """ 

160 # Local imports: keep this module import-safe from anywhere (views, commands). 

161 from accounts.models import ArchivedUser 

162 from catalog.models import CatalogItem 

163 from commerce.models import CartItem, Order 

164 

165 candidates = _eligibility_queryset(days=days, force=force) 

166 

167 moved = 0 

168 for user in candidates: 

169 closed_id = str(user.pk) 

170 try: 

171 # 1) Pseudonymized cold tombstone FIRST (idempotent via closed_account_id). 

172 ArchivedUser.from_user(user) 

173 

174 # 2) Detach buyers' carts that reference THIS seller's listings, 

175 # so cascading the listings away later can't dangle a CartItem. 

176 # (Verbatim from the original ProfileView.destroy.) 

177 for item in CatalogItem.objects.filter(seller=user): 

178 CartItem.objects.filter(item=item).update( 

179 item=None, 

180 catalog_item_id=str(item.pk), 

181 item_name=item.name, 

182 store_name=item.store_name, 

183 unavailable_reason=( 

184 "This item is unavailable because the seller’s account was deleted." 

185 ), 

186 seller_account_deleted=True, 

187 ) 

188 

189 # 3) Pseudonymize this user's orders: keep the order RECORD 

190 # (totals/items/dates/status) but drop the personal link + PII. 

191 # (Verbatim from the original ProfileView.destroy.) 

192 for order in Order.objects.filter(user=user): 

193 order.user = None 

194 order.recipient_name = '' 

195 order.contact_number = '' 

196 order.recipient_email = '' 

197 order.shipping_address = '' 

198 order.save(update_fields=[ 

199 'user', 'recipient_name', 'contact_number', 

200 'recipient_email', 'shipping_address', 

201 ]) 

202 

203 # 4) Remove the live profile. Cascades delete the user's own 

204 # presets / outfits / cart / remaining listings / ratings / 

205 # fitting-studio surveys, exactly as the original hard-delete 

206 # did. Orders survived step 3 (user=None) and cart rows survived 

207 # step 2 (item=None). 

208 user.delete() 

209 moved += 1 

210 # FR-1.4: never write the deleted account's email into logs. 

211 logger.info( 

212 "Cold-archived user %s%s.", 

213 closed_id, 

214 " via --force" if force else f" after the {days}-day grace window", 

215 ) 

216 except Exception: # noqa: BLE001 - one bad row must not stall the sweep 

217 logger.exception("Failed to cold-archive user %s; skipping.", closed_id) 

218 

219 if moved: 

220 logger.info("Prune sweep relocated %s archived user(s) to cold storage.", moved) 

221 return moved
