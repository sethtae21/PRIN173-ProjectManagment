6class StrictStarField(serializers.IntegerField): 

7 """ 

8 STRICT 1..5 integer stars (SRS FR-7.5 star-only baseline). 

9 

10 DRF's IntegerField happily coerces "5", 5.0 and True into 5. The baseline 

11 requires an actual integer, so every non-int (including bool, which is an 

12 int subclass in Python) is rejected with 400 instead of being coerced. 

13 """ 

14 

15 def to_internal_value(self, data): 

16 if isinstance(data, bool) or not isinstance(data, int): 

17 self.fail('invalid') 

18 return super().to_internal_value(data)

21class RatingSerializer(serializers.ModelSerializer): 

22 """ 

23 STAR-ONLY payload: { target_type, target_id, score (int 1..5), source }. 

24 `score` and `source` are REQUIRED. There is no vote/like/dislike field. 

25 

26 HARDENING (defense-in-depth): legacy `vote` / `like` / `dislike` keys in 

27 the request body are EXPLICITLY REJECTED (400), not silently dropped. 

28 Without this guard, a stale client sending `{"score": 5, "vote": "like"}` 

29 would still save the score and the vote would vanish — masking the 

30 client-side drift from the terminal. 

31 """ 

32 id = serializers.CharField(read_only=True) 

33 user = serializers.CharField(read_only=True, source='user_id') 

34 

35 score = StrictStarField(min_value=1, max_value=5) 

36 source = serializers.ChoiceField(choices=Rating.SOURCE_CHOICES) 

37 

38 class Meta: 

39 model = Rating 

40 fields = ['id', 'user', 'target_type', 'target_id', 'score', 

41 'source', 'created_at', 'updated_at'] 

42 read_only_fields = ['user', 'created_at', 'updated_at']

46 legacy = {'vote', 'like', 'dislike'}.intersection(data.keys())
