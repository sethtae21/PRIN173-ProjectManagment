1""" 

2Management command: FR-1.4 / RA 10173 backfill for ALREADY-ARCHIVED records. 

3 

4Changing future archive writes does not fix legacy accounts_archiveduser 

5documents that were copied intact by older builds (or sanitized under the old 

6placeholder/null scheme). This command pseudonymizes them in place via the 

7live helper: keyed pseudonymous email, pseudonymous display name 

8("Archived <32hex>"), field-specific MOB-/ADDR-/POST-/STORE- tokens for 

9contact/address/postal/store identity (never null), empty password, cleared 

10behavioral timestamps. 

11 

12Idempotent: a second run reports pseudonymized=0 / already_clean=N and writes 

13nothing. --dry-run computes the same counts without saving. 

14 

15Usage: 

16 python manage.py pseudonymize_archived_users --dry-run 

17 python manage.py pseudonymize_archived_users 

18 python manage.py pseudonymize_archived_users # again -> already_clean=N 

19""" 

20from django.core.management.base import BaseCommand 

21 

22from accounts.pseudonymization import sanitize_existing_archives 

23 

24 

25class Command(BaseCommand): 

26 help = ( 

27 "Idempotent FR-1.4 backfill: pseudonymize legacy accounts_archiveduser " 

28 "rows (keyed tokens for every direct identifier — never null or the " 

29 "original; password cleared). Safe to re-run; --dry-run reports " 

30 "without writing." 

31 ) 

32 

33 def add_arguments(self, parser): 

34 parser.add_argument( 

35 '--dry-run', 

36 action='store_true', 

37 help="Report what would be pseudonymized without writing anything.", 

38 ) 

39 

40 def handle(self, *args, **options): 

41 apply = not options['dry_run'] 

42 stats = sanitize_existing_archives(apply=apply) 

43 prefix = '[dry-run] ' if not apply else '' 

44 self.stdout.write(self.style.SUCCESS( 

45 f"{prefix}scanned={stats['scanned']} " 

46 f"pseudonymized={stats['pseudonymized']} " 

47 f"already_clean={stats['already_clean']}" 

48 ))
