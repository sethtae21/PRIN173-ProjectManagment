1""" 

2accounts/survey_dedupe.py 

3Idempotent (user, session_id) dedupe for FittingStudioSurvey. 

4 

5Called by migration 0011 BEFORE the unique index is created, and safe to call 

6again: a failed migrate on MongoDB is not rolled back, so this helper may run 

7more than once over partially-deduped data. 

8 

9CALL CONTRACT — every shape below is supported. Do NOT narrow this signature: 

10migration 0011 calls it with `apply=True`, and narrowing it to 

11(apps, schema_editor) breaks test-database creation for the whole project. 

12 dedupe_fitting_studio_surveys(apply=True) # migration 0011 

13 dedupe_fitting_studio_surveys(apply=False) # dry run 

14 dedupe_fitting_studio_surveys(False) # legacy positional bool 

15 dedupe_fitting_studio_surveys(apps, schema_editor) # RunPython style 

16 dedupe_fitting_studio_surveys(SurveyModel) # explicit model class 

17 dedupe_fitting_studio_surveys() # live model, apply 

18 dedupe_fitting_studio_surveys(dry_run=True) # alias for apply=False 

19 

20MongoDB-safe: grouping happens in Python rather than .values().annotate(Count), 

21so it does not depend on backend aggregation support. 

22 

23Two-phase preservation: the NEWEST row per key is kept (by submitted_at, the 

24model's real timestamp), and any field the keeper left null is filled from an 

25older row before that row is deleted, so a phase-1 answer stored on an older 

26duplicate is never lost. With apply=False nothing is saved or deleted; the 

27in-memory merge still runs so the reported counts are accurate. 

28""" 

29from __future__ import annotations 

30 

31# submitted_at is this model's timestamp (there is no created_at/updated_at); 

32# the other names are kept so the helper still works if those are added later. 

33ORDER_FIELDS = ('submitted_at', 'updated_at', 'created_at', 'id') 

34PROTECTED = { 

35 'id', 'pk', 'user', 'user_id', 'session_id', 

36 'submitted_at', 'created_at', 'updated_at', 

37} 

38 

39 

40def _resolve_model(apps): 

41 if apps is not None and hasattr(apps, 'get_model'): # RunPython(apps, editor) 

42 return apps.get_model('accounts', 'FittingStudioSurvey') 

43 if isinstance(apps, type): # called with the model class 

44 return apps 

45 from accounts.models import FittingStudioSurvey # apply=/no-args call 

46 return FittingStudioSurvey

49def dedupe_fitting_studio_surveys(apps=None, schema_editor=None, apply=True, *, dry_run=False): 

50 # Tolerate the legacy positional form dedupe_fitting_studio_surveys(False). 

51 if isinstance(apps, bool): 

52 apply = apps 

53 apps = None 

54 if dry_run: 

55 apply = False 

56 

57 Survey = _resolve_model(apps) 

58 names = {f.name for f in Survey._meta.concrete_fields} 

59 order = [f'-{n}' for n in ORDER_FIELDS if n in names] 

60 payload_fields = sorted(names - PROTECTED) 

61 

62 groups = {} 

63 for row in Survey.objects.all().order_by(*order).iterator(): 

64 key = (str(getattr(row, 'user_id', '')), row.session_id) 

65 groups.setdefault(key, []).append(row) 

66 

67 stats = { 

68 'scanned': 0, 

69 'duplicate_groups': 0, 

70 'merged_fields': 0, 

71 'deleted': 0, 

72 'applied': bool(apply), 

73 } 

74 for rows in groups.values(): 

75 stats['scanned'] += len(rows) 

76 if len(rows) < 2: 

77 continue 

78 stats['duplicate_groups'] += 1 

79 keep, *dupes = rows # newest first because of order_by 

80 changed = [] 

81 for older in dupes: 

82 for field in payload_fields: 

83 if getattr(keep, field) is None and getattr(older, field) is not None: 

84 setattr(keep, field, getattr(older, field)) 

85 changed.append(field) 

86 if changed: 

87 stats['merged_fields'] += len(set(changed)) 

88 if apply: 

89 keep.save(update_fields=sorted(set(changed))) 

90 for older in dupes: 

91 stats['deleted'] += 1 

92 if apply: 

93 older.delete() 

94 return stats
