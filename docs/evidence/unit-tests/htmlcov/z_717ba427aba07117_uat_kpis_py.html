1"""ratings/uat_kpis.py — Charter-aligned UAT KPI engine (single source of truth). 

2 

3Traceability: Project Charter §3.2 (objectives + acceptance criteria), SRS §1.2, 

4Scope Criteria 1–11, WBS 1.4.x UAT plan, KAN-89 evidence wiring. 

5 

6Rules baked into this module: 

7 * Charter numbers are BASELINED acceptance criteria. They are never 

8 "proposed", never retuned to fit captured data, and any edit is a 

9 baselined change (adviser/CCB), guarded by CharterThresholdLockTests. 

10 * Percent-of-eligible-sessions KPIs (OBJ-01/06) use paired before/after 

11 answers only; unpaired answers never enter the denominator. 

12 * OBJ-05 reads in-app ratings tagged source='recommendation_panel' only. 

13 * OBJ-07 counts trusted server-side commerce events, never client claims 

14 and never manual POSTs (the manual event endpoint was removed so the 

15 evidence cannot be falsified). 

16 * Objectives whose charter evidence is observed/measured (OBJ-02/03/04/08/09) 

17 are reported as manual-evidence rows. Subjective survey answers are 

18 exported only as "Context only" rows and are never substituted. 

19 

20STAR-ONLY BASELINE (SRS FR-7.5 / Scope Criteria 9-11): 

21 Approval is expressed ONLY as an integer star score >= APPROVAL_MIN_SCORE 

22 (4). The former vote='like' approval path was retired together with the 

23 Rating.vote column (ratings migration 0005). Every locked charter threshold 

24 is unchanged; only the *expression* of approval moved to stars. 

25 

26Keep the questionnaire constants in sync with frontend/src/services/uat.js. 

27"""

143def _json_notes(raw): 

144 if not raw: 

145 return {} 

146 if isinstance(raw, dict): 

147 return raw 

148 try: 

149 parsed = json.loads(raw) 

150 except (TypeError, ValueError): 

151 return {} 

152 return parsed if isinstance(parsed, dict) else {}

155def _paired_values(question_key): 

156 """Paired before/after answers keyed by (session PK, product id str). 

157 

158 Only keys answered in BOTH phases are returned: an unpaired answer cannot 

159 evidence a within-session change and stays out of the eligible 

160 denominator. `value == 0` is a legal answer and is kept (value__isnull). 

161 """ 

162 collected = {} 

163 for response in UATResponse.objects.filter( 

164 question_key=question_key, 

165 value__isnull=False, 

166 ).values('session_id', 'product_id', 'phase', 'value'): 

167 key = (response['session_id'], str(response['product_id'])) 

168 collected.setdefault(key, {})[response['phase']] = int(response['value']) 

169 return {key: phases for key, phases in collected.items() 

170 if PHASE_BEFORE in phases and PHASE_AFTER in phases}

173def _confidence_improved(pair): 

174 """Criterion 9 per-question condition: gain ≥ +1 OR post ≥ 4/5.""" 

175 return (pair[PHASE_AFTER] - pair[PHASE_BEFORE] >= CONFIDENCE_GAIN_MIN 

176 or pair[PHASE_AFTER] >= CONFIDENCE_POST_PASS)

179def _rating_approves(row): 

180 """STAR-ONLY approval (SRS FR-7.5): an integer score >= 4 approves. 

181 

182 CHANGED (audit fix): the former `vote == 'like'` path is GONE — the vote 

183 column was removed by ratings migration 0005, so a like can no longer 

184 masquerade as a star approval in any charter KPI. 

185 """ 

186 score = row.get('score') 

187 return score is not None and int(score) >= APPROVAL_MIN_SCORE

190def _rate_summary(numerator, denominator, threshold_pct, evidence_reference): 

191 percentage = (numerator / denominator) * PERCENTAGE_SCALE if denominator else 0.0 

192 return { 

193 'actual_result': f'{percentage:.2f}%', 

194 'sample_size': denominator, 

195 'numerator': str(numerator), 

196 'denominator': str(denominator), 

197 'status': STATUS_MET if percentage >= threshold_pct else STATUS_BELOW, 

198 'evidence_reference': evidence_reference, 

199 }

202def _at_least(threshold_pct): 

203 return f'≥ {threshold_pct:g}%'

209def _obj01_fit_style_summary(): 

210 """Criterion 9: % of ELIGIBLE try-on sessions where fit AND style 

211 confidence improved AND the fit visualization was approved (>= 4 stars). 

212 Fit ratings join to sessions by (user, product) — UATSessionCreateView 

213 persists product_id — and must be tagged source='fit_visualization' / 

214 target_type='fit'.""" 

215 fit_pairs = _paired_values('fit_confidence') 

216 style_pairs = _paired_values('style_confidence') 

217 eligible_keys = sorted(set(fit_pairs) & set(style_pairs)) 

218 evidence = ('Pre/post fit_confidence + style_confidence answers joined to ' 

219 'fit_visualization in-app STAR ratings (FR-7.5 / KAN-89 / KAN-101). ' 

220 f'Approval = score ≥ {APPROVAL_MIN_SCORE}/5.') 

221 if not eligible_keys: 

222 return {**_not_run(), 

223 'evidence_reference': f'{evidence} No eligible sessions captured yet.'} 

224 

225 session_user = {(session.pk, str(session.product_id)): session.user_id 

226 for session in UATSession.objects.all()} 

227 

228 # STAR-ONLY: read score only; the vote column no longer exists. 

229 fit_approvals = set() 

230 for row in Rating.objects.filter( 

231 target_type='fit', 

232 source=SOURCE_FIT_VISUALIZATION, 

233 ).values('user_id', 'target_id', 'score'): 

234 if _rating_approves(row): 

235 fit_approvals.add((row['user_id'], str(row['target_id']))) 

236 

237 met = 0 

238 for key in eligible_keys: 

239 fit_ok = _confidence_improved(fit_pairs[key]) 

240 style_ok = _confidence_improved(style_pairs[key]) 

241 user_id = session_user.get(key) 

242 approved = user_id is not None and (user_id, key[1]) in fit_approvals 

243 if fit_ok and style_ok and approved: 

244 met += 1 

245 

246 return _rate_summary(met, len(eligible_keys), 

247 CHARTER_THRESHOLDS['OBJ-01'], evidence)

250def _obj05_recommendation_approval_summary(): 

251 """Criterion 10: % of recorded recommendation_panel ratings approving. 

252 Product-page / fit ratings and all survey questions are excluded. 

253 STAR-ONLY: approval = score >= APPROVAL_MIN_SCORE.""" 

254 rows = list(Rating.objects.filter( 

255 target_type='item', 

256 source=SOURCE_RECOMMENDATION_PANEL, 

257 ).values('score')) 

258 evidence = ('Rating rows with source=recommendation_panel (FR-7.5 / KAN-101). ' 

259 f'Approval = score ≥ {APPROVAL_MIN_SCORE}/5 (stars only).') 

260 if not rows: 

261 return {**_not_run(), 

262 'evidence_reference': f'{evidence} No ratings recorded yet.'} 

263 approvals = sum(1 for row in rows if _rating_approves(row)) 

264 return _rate_summary(approvals, len(rows), CHARTER_THRESHOLDS['OBJ-05'], evidence)

267def _obj06_purchase_decision_summary(): 

268 """Criterion 11: % of ELIGIBLE sessions with purchase confidence STRICTLY 

269 increased AND return intent STRICTLY decreased vs the before baseline.""" 

270 confidence_pairs = _paired_values('purchase_confidence') 

271 return_pairs = _paired_values('return_intent') 

272 eligible_keys = sorted(set(confidence_pairs) & set(return_pairs)) 

273 evidence = ('Paired pre/post purchase_confidence and return_intent responses ' 

274 '(KAN-89). Session passes only when confidence strictly increased ' 

275 'AND return intent strictly decreased.') 

276 if not eligible_keys: 

277 return {**_not_run(), 

278 'evidence_reference': f'{evidence} No eligible sessions captured yet.'} 

279 met = 0 

280 for key in eligible_keys: 

281 confidence = confidence_pairs[key] 

282 returns = return_pairs[key] 

283 if (confidence[PHASE_AFTER] > confidence[PHASE_BEFORE] 

284 and returns[PHASE_AFTER] < returns[PHASE_BEFORE]): 

285 met += 1 

286 return _rate_summary(met, len(eligible_keys), 

287 CHARTER_THRESHOLDS['OBJ-06'], evidence)

290def _obj07_checkout_completion_summary(): 

291 """Criterion 8: completed mock checkouts ÷ carts created during UAT. 

292 

293 Events are written server-side (record_uat_commerce_event) on the 

294 tester's own ACTIVE session; try-on-only sessions never enter the 

295 denominator. The per-session boolean flags are an EXACT cart count in 

296 this system: the cart is a per-user singleton (Cart.objects.get_or_create) 

297 and a session flips to 'completed' the moment its checkout succeeds, so 

298 one session can contribute at most one created cart and one completed 

299 checkout — and post-checkout cart events are ignored by the hook. 

300 """ 

301 carts = 0 

302 completed = 0 

303 for session in UATSession.objects.all(): 

304 notes = _json_notes(session.notes) 

305 if notes.get(EVENT_CART_CREATED) is not True: 

306 continue 

307 carts += 1 

308 if notes.get(EVENT_CHECKOUT_COMPLETED) is True: 

309 completed += 1 

310 evidence = ('Server-side UAT commerce events (cart_created / checkout_completed) ' 

311 'recorded on active shopper UAT sessions (KAN-89).') 

312 if carts == 0: 

313 return {**_not_run(), 

314 'evidence_reference': f'{evidence} No UAT carts recorded yet.'} 

315 return _rate_summary(completed, carts, CHARTER_THRESHOLDS['OBJ-07'], evidence)

324def _rating_mean_summary(question_key): 

325 values = [int(v) for v in UATResponse.objects.filter( 

326 question_key=question_key, value__isnull=False, 

327 ).values_list('value', flat=True)] 

328 if not values: 

329 return {**_not_run(), 'evidence_reference': _CONTEXT_EVIDENCE} 

330 mean = sum(values) / len(values) 

331 return {'actual_result': f'{mean:.2f} / 5', 'sample_size': len(values), 

332 'numerator': '', 'denominator': '', 'status': STATUS_CALCULATED, 

333 'evidence_reference': _CONTEXT_EVIDENCE}

336def _yes_no_summary(question_key): 

337 values = [int(v) for v in UATResponse.objects.filter( 

338 question_key=question_key, value__isnull=False, 

339 ).values_list('value', flat=True)] 

340 if not values: 

341 return {**_not_run(), 'evidence_reference': _CONTEXT_EVIDENCE} 

342 yes = sum(1 for v in values if v == YES_NO_VALUE_YES) 

343 pct = (yes / len(values)) * PERCENTAGE_SCALE 

344 return {'actual_result': f'{pct:.2f}% Yes', 'sample_size': len(values), 

345 'numerator': str(yes), 'denominator': str(len(values)), 

346 'status': STATUS_CALCULATED, 'evidence_reference': _CONTEXT_EVIDENCE}

467def _apply_uat_event(session, event): 

468 from django.utils import timezone 

469 

470 notes = _json_notes(session.notes) 

471 notes[event] = True 

472 event_log = notes.get('event_log') 

473 if not isinstance(event_log, list): 

474 event_log = [] 

475 event_log.append({'event': event, 'at': timezone.now().isoformat()}) 

476 notes['event_log'] = event_log 

477 session.notes = json.dumps(notes) 

478 if event == EVENT_CHECKOUT_COMPLETED: 

479 session.status = 'completed' 

480 session.save()

483def record_uat_commerce_event(user, event): 

484 """Server-side OBJ-07 hook called by commerce views (cart add_item and the 

485 successful checkout path only). 

486 

487 Records the event on the user's most recent ACTIVE UAT session. 

488 Owner-scoped, append-only, and never raises: UAT bookkeeping must not 

489 be able to break a cart or checkout. Returns True if recorded. 

490 

491 Because the session flips to 'completed' when checkout succeeds, later 

492 cart events for that session are ignored — one session can therefore 

493 never inflate the created-cart denominator. 

494 """ 

495 try: 

496 if event not in UAT_COMMERCE_EVENTS: 

497 return False 

498 if user is None or not getattr(user, 'is_authenticated', False): 

499 return False 

500 session = (UATSession.objects.filter(user=user, status='active') 

501 .order_by('-started_at').first()) 

502 if session is None: 

503 return False 

504 _apply_uat_event(session, event) 

505 return True 

506 except Exception: # noqa: BLE001 — commerce flow must never break 

507 logger.exception('UAT commerce event recording failed (event=%s); ' 

508 'commerce flow unaffected.', event) 

509 return False
